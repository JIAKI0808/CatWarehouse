================================================================================
  目录结构
================================================================================

CwServer/
├── catWarehouse_server.py          # FastAPI 入口，lifespan，CORS
├── requirements.txt                # Python 依赖
├── .env.example                    # 环境变量示例
│
├── core/                           # 基础设施层
│   ├── config.py                   #   Settings 配置管理
│   ├── database.py                 #   SQLAlchemy 异步引擎
│   └── logging.py                  #   日志配置
│
├── models/                         # 数据库模型层
│   ├── __init__.py                 #   Base + 导入所有模型
│   ├── category.py                 #   Category 大类表
│   ├── sub_category.py             #   SubCategory 细分类表
│   └── specific_item.py            #   SpecificItem 具体条目表
│
├── schemas/                        # Pydantic 校验层
│   ├── category.py                 #   CategoryCreate / Response
│   ├── sub_category.py             #   SubCategoryCreate / Response
│   └── specific_item.py            #   SpecificItemCreate / Update / Response
│
├── api/                            # 路由层
│   └── router.py                   #   所有 API 端点
│
└── dataManager/                    # 抽象工厂层
    ├── base.py                     #   抽象基类
    ├── factory.py                  #   工厂注册与获取
    └── inventory/                  #   具体工厂：商品/库存
        ├── factory.py              #     InventoryManagerFactory
        ├── item.py                 #     Item 数据产品
        ├── validator.py            #     ItemValidator 验证器
        └── repository.py           #     ItemRepository 存储仓库
